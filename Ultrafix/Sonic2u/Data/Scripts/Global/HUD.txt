private function HUD_Draw_Standard

#platform: USE_ORIGINS
	// temp4 will tell whether to draw lives or coins
	temp4 = game.coinMode
	
	// draw at an offset of 0 by default
	temp6 = 0
	
	if game.playMode == BOOT_PLAYMODE_MISSION
		// stop drawing the HE2 HUD when the player dies
		if game.missionCondition == MISSION_CONDITION_FAIL
			game.hudEnable = HUDENABLE_OFF
		else
			game.hudEnable = HUDENABLE_ON
		end if
		
		// always display coins in mission mode
		temp4 = true
		
		// draw at an offset since score is removed
		temp6 = -12
	else
		if game.playMode == BOOT_PLAYMODE_BOSSRUSH
			// always display lives in boss rush
			temp4 = false
			
			// draw at an offset since score is removed
			temp6 = -12
		else
			if options.vsMode == true
				// always display lives in 2PVS
				temp4 = false
			end if
		end if
	end if
#endplatform
	
	// Draw the base SCORE/TIME/RINGS text
	DrawSpriteScreenXY(10, 16, 9)
	
#platform: USE_ORIGINS
	if screen.currentID == 0
		temp2 = object.vsP1Finished
	else
		temp2 = object.vsP2Finished
	end if
	if temp2 == true
		if screen.currentID == 0
			temp2 = object.vsP1Minutes
			temp3 = object.vsP1Seconds
		else
			temp2 = object.vsP2Minutes
			temp3 = object.vsP2Seconds
		end if
	else
		temp2 = stage.minutes
		temp3 = stage.seconds
	end if
	
	CheckGreater(temp2, 9)
	temp5 = checkResult
	CheckNotEqual(object.timeOver, false)
	checkResult |= temp5
	
	if checkResult != false
		temp2 = 9
		temp3 = 59
	end if
	
	if temp2 == 9
#endplatform
#platform: USE_STANDALONE
	if stage.minutes == 9
#endplatform
		CheckEqual(game.coinMode, false)
		checkResult |= options.vsMode
		if checkResult == true
			// Overlay the red TIME text
			if object.flashingTimer > 7
				temp0 = 25
				temp0 += temp6
				DrawSpriteScreenXY(11, 16, temp0)
			end if
		end if
	end if

	// Overlay the red RINGS text if needed as well
#platform: USE_STANDALONE
	if player[0].rings == 0
#endplatform
#platform: USE_ORIGINS
	if player[currentPlayer].rings == 0
#endplatform
		if object.flashingTimer > 7
			temp0 = 41
			temp0 += temp6
			DrawSpriteScreenXY(12, 16, temp0)
		end if
	end if

#platform: USE_STANDALONE
	DrawNumbers(0, 103, 9, player.score, 6, 8, false)
#endplatform
#platform: USE_ORIGINS
	// only draw score if offset is 0, that means we're not in mission/boss rush!
	if temp6 == 0
		arrayPos0 = GLOBAL_PLAYERSCORE // VarName[player.score]
		arrayPos0 += currentPlayer
		DrawNumbers(0, 103, 9, global[arrayPos0], 7, 8, false)
	end if
#endplatform

	temp0 = 25
	temp0 += temp6

	if options.gameMode < MODE_TIMEATTACK
		// Normal game - only minutes and seconds are shown
		DrawSpriteScreenXY(14, 77, temp0)
	else
		// Time attack - draw milliseconds alongside of the other values
		DrawSpriteScreenXY(13, 77, temp0)
		
		if screen.currentID == 0
			temp1 = object.vsP1Finished
		else
			temp1 = object.vsP2Finished
		end if
		if temp1 == true
			if screen.currentID == 0
				temp1 = object.vsP1MilliSecs
			else
				temp1 = object.vsP2MilliSecs
			end if
		else
			temp1 = stage.milliSeconds
		end if
		
		DrawNumbers(0, 103, temp0, temp1, 2, 8, true)
	end if

	// Rest of the time display
#platform: USE_STANDALONE
	DrawNumbers(0, 90, temp0, stage.seconds, 2, 8, true)
	DrawNumbers(0, 66, temp0, stage.minutes, 1, 8, true)
	
	temp0 = 41
	temp0 += temp6
	
	DrawNumbers(0, 90, temp0, player[0].rings, 3, 8, false)
#endplatform
#platform: USE_ORIGINS
	DrawNumbers(0, 90, temp0, temp3, 2, 8, true)
	DrawNumbers(0, 66, temp0, temp2, 1, 8, true)
	
	temp0 = 41
	temp0 += temp6
	
	DrawNumbers(0, 90, temp0, player[currentPlayer].rings, 3, 8, false)
#endplatform

	if stage.debugMode == true
		if game.playMode != BOOT_PLAYMODE_MISSION
			// Player position display

			// XPos display
#platform: USE_STANDALONE
			temp0 = player[0].xpos
#endplatform
#platform: USE_ORIGINS
			temp0 = player[currentPlayer].xpos
#endplatform
			temp0 >>= 16
			Abs(temp0)
			temp1 = screen.xsize
			temp1 -= 20
			DrawNumbers(0, temp1, 13, temp0, 5, 8, true)

			// YPos display
#platform: USE_STANDALONE
			temp0 = player[0].ypos
#endplatform
#platform: USE_ORIGINS
			temp0 = player[currentPlayer].ypos
#endplatform
			temp0 >>= 16
			Abs(temp0)
			DrawNumbers(0, temp1, 29, temp0, 5, 8, true)

			// Move 42 pixels to the left
			temp1 -= 42

			// Draw the "X:" and "Y:" icons

			// Use red variant if player is in a negative pos
#platform: USE_STANDALONE
			if player[0].xpos >= 0
#endplatform
#platform: USE_ORIGINS
			if player[currentPlayer].xpos >= 0
#endplatform
				DrawSpriteScreenXY(18, temp1, 15)
			else
				DrawSpriteScreenXY(36, temp1, 15)
			end if

#platform: USE_STANDALONE
			if player[0].ypos >= 0
#endplatform
#platform: USE_ORIGINS
			if player[currentPlayer].ypos >= 0
#endplatform
				DrawSpriteScreenXY(19, temp1, 31)
			else
				DrawSpriteScreenXY(37, temp1, 31)
			end if
		end if
	end if
	
#platform: USE_STANDALONE
	// Lives icon
	temp0 = stage.playerListPos
	temp0 += 15
	DrawSpriteScreenXY(temp0, 16, 216)

	// Character text
	temp0 += 6
	DrawSpriteScreenXY(temp0, 33, 217)

	// X between icon and number
	DrawSpriteScreenXY(20, 38, 226)

	// The actual player lives number
	DrawNumbers(24, 57, 224, player.lives, 2, 8, false)
#endplatform

#platform: USE_ORIGINS
	if temp4 == false
		// Get the character sprites
		if currentPlayer == 0 // If this is Player 1
			temp0 = stage.playerListPos
		else
			temp0 = vs.player2Type
		end if

		if temp0 == PLAYER_AMY
			temp0 = 40
			temp1 = 41
		else
			temp0 += 15
			temp1 = temp0
			temp1 += 6
		end if
		
		DrawSpriteScreenXY(temp0, 16, 216) // Life icon
		DrawSpriteScreenXY(temp1, 33, 217) // Character name tag

		// X between icon and number
		DrawSpriteScreenXY(20, 37, 226)
		
		// The actual player lives number
		arrayPos0 = GLOBAL_PLAYERLIVES // VarName[player.lives]
		arrayPos0 += currentPlayer
		DrawNumbers(24, 56, 224, global[arrayPos0], 2, 8, false)
	else
		// Draw the Coins counter
		DrawSpriteScreenXY(38, 16, 216)
		DrawSpriteScreenXY(39, 33, 217)
		DrawSpriteScreenXY(20, 35, 226)
		DrawNumbers(24, 57, 224, game.coinCount, 3, 8, false)
	end if
#endplatform
end function


private function HUD_Draw_Mobile
	DrawSpriteScreenXY(10, 17, 9)

	if stage.minutes == 9
		if object.flashingTimer > 7
			DrawSpriteScreenXY(11, 17, 25)
		end if
	end if

	if player[0].rings == 0
		if object.flashingTimer > 7
			DrawSpriteScreenXY(12, 17, 41)
		end if
	end if

	DrawNumbers(0, 104, 9, player.score, 6, 8, false)

	if options.gameMode < MODE_TIMEATTACK
		DrawSpriteScreenXY(14, 67, 25)
	else
		DrawSpriteScreenXY(13, 67, 25)
		DrawNumbers(0, 104, 25, stage.milliSeconds, 2, 8, true)
	end if

	DrawNumbers(0, 90, 25, stage.seconds, 2, 8, true)
	DrawNumbers(0, 66, 25, stage.minutes, 1, 8, true)
	DrawNumbers(0, 90, 41, player[0].rings, 3, 8, false)

	if player[0].type == TypeName[Debug Mode]
		// Player position display, only show it when the player is in Debug Mode (as opposed to Standard's, where it's drawn if Debug Mode is active at all)

		temp0 = player[0].xpos
		temp0 >>= 16
		Abs(temp0)
		temp1 = screen.xsize
		temp1 -= 16
		DrawNumbers(0, temp1, 25, temp0, 5, 8, true)

		temp0 = player[0].ypos
		temp0 >>= 16
		Abs(temp0)
		DrawNumbers(0, temp1, 41, temp0, 5, 8, true)

		temp1 -= 42
		if player[0].xpos >= 0
			DrawSpriteScreenXY(18, temp1, 27)
		else
			DrawSpriteScreenXY(36, temp1, 27)
		end if

		if player[0].ypos >= 0
			DrawSpriteScreenXY(19, temp1, 43)
		else
			DrawSpriteScreenXY(37, temp1, 43)
		end if

		DrawSpriteScreenXY(34, screen.xcenter, 16)
		DrawSpriteScreenXY(35, screen.xcenter, 16)
	end if

	temp1 = screen.xsize
	temp1 -= 64
	
	temp0 = stage.playerListPos
	
#platform: USE_STANDALONE
	temp0 += 15
	DrawSpriteScreenXY(temp0, temp1, 8)
#endplatform

#platform: USE_ORIGINS
	// You never see the mobile HUD in Origins yet they updated it in Plus LOL
	// It wasn't even properly updated, since temp1 is being overwritten the x position of the sprites are now broken
	if temp0 == PLAYER_AMY
		temp0 = 40
		temp1 = 41
	else
		temp0 += 15
		temp1 = temp0
		temp1 += 6
	end if
	DrawSpriteScreenXY(temp0, temp1, 8)
#endplatform

	temp1 += 17
	temp0 += 6
	DrawSpriteScreenXY(temp0, temp1, 9)

	temp1 += 5
	DrawSpriteScreenXY(20, temp1, 18)

	temp1 += 18
	DrawNumbers(24, temp1, 16, player.lives, 2, 8, false)
end function

event ObjectDraw
#platform: USE_ORIGINS
	if game.timeOver != false
		object.timeOver = true
	end if
#endplatform
	
	// Update flashing red texts
#platform: USE_STANDALONE
	if player[0].rings == 0
		object.flashingTimer++

		if object.flashingTimer == 16
			object.flashingTimer = 0
		end if
	else
		if stage.minutes == 9
			object.flashingTimer++

			if object.flashingTimer == 16
				object.flashingTimer = 0
			end if
		end if
	end if
#endplatform

#platform: USE_ORIGINS
	currentPlayer = screen.currentID
	
	// Progress HUD flashing
	CheckEqual(player[currentPlayer].rings, 0)
	temp0 = checkResult
	CheckEqual(stage.minutes, 9)
	temp0 |= checkResult
	temp0 |= object.timeOver
	if temp0 == true
		object.flashingTimer = oscillation
		object.flashingTimer &= 15
	end if
#endplatform
	
	// cap the player's score
	if player.score > 9999999
		player.score = 9999999
	end if
	if player.score2 > 9999999
		player.score2 = 9999999
	end if
	
	// cap the player's lives
	if player.lives > 99
		player.lives = 99
	end if
	if player.lives2 > 99
		player.lives2 = 99
	end if

	SetActivePalette(0, 0, 256)		// don't use the underwater palette!
	
	CallFunction(object.drawFunction)
end event
